# Filter and reshape the data to long format
production_long <- raw_materials_data %>%
  filter(country %in% major_nations) %>%
  select(country, coal_produced, iron_produced, oil_produced) %>%
  pivot_longer(
    cols = c(coal_produced, iron_produced, oil_produced),
    names_to = "material",
    values_to = "amount"

# Calculate mean annual production per country and material
mean_production <- production_long %>%
  group_by(country, material) %>%

# Filter only countries in the grouping
grouped_data <- data_raw_materials_with_allies %>%
  filter(country %in% names(country_groups)) %>%

# Convert to long format and summarise mean production
grouped_long <- grouped_data %>%

2.2: Comparing tank production of Major Nations (figure 3)

tank_summary <- tank_data %>%
  filter(!is.na(tanks_produced)) %>%
  mutate(year = as.factor(year)) %>%
  group_by(country, year) %>%
  summarise(total_tanks = sum(tanks_produced, na.rm = TRUE), .groups = "drop")

tank_summary <- tank_summary %>%
  group_by(country) %>%
  mutate(total_all_years = sum(total_tanks)) %>%
  ungroup() %>%
  mutate(country = reorder(country, -total_all_years))

year_colors <- c(
  "1939" = "grey",
  "1940" = "darkgrey",
  "1941" = "lightgrey",
  "1942" = "darkred",
  "1943" = "steelblue",
  "1944" = "lightblue",
  "1945" = "pink"
)
total_per_country <- tank_summary %>%
  group_by(country) %>%
  summarise(total_all_years = sum(total_tanks), .groups = "drop") %>%
  mutate(country = factor(country, levels = levels(tank_summary$country)))

ggplot(tank_summary, aes(x = country, y = total_tanks, fill = year)) +
  geom_col() +
  geom_text(aes(label = comma(total_tanks)),
            position = position_stack(vjust = 0.5),
            size = 2.5, color = "black") +
  geom_text(
    data = total_per_country,
    aes(x = country, y = total_all_years, label = comma(total_all_years)),
    vjust = -0.5, size = 4, fontface = "bold",
    inherit.aes = FALSE
  ) +
  scale_fill_manual(values = year_colors) + 
  scale_y_continuous(labels = comma_format()) +
  labs(
    title = "Tank Production by Country and Year",
    x = "Nation (note missing data for Germany/Britain in 1945)",
    y = "Tank Production of 1939-45",
    fill = "Year"
  ) +
  theme_minimal()
